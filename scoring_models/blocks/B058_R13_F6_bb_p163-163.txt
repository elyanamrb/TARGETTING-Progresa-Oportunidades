Región 13. Mixteca - Costa - Chatina
Fase 6
Modelo bb

**discriminante con modelo nuevo

DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES= depen ninos ocupj_1 esc piso_1 ej aguaba_1 trab151 noasi1 pared_1
refri vehi estgas tv
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS .41 .59
/STATISTICS=RAW TABLE
/CLASSIFY=NONMISSING POOLED .

***puntajes región 13

COMPUTE bbDEPEN = -31*DEPEN .
COMPUTE bbNINOS = -20*NINOS .
COMPUTE bbOCUPJ = -41*OCUPJ_1 .
COMPUTE bbESC = 8*ESC .
COMPUTE bbPISO = -2*PISO_1 .
COMPUTE bbEDAD = -1*EJ .
COMPUTE bbAGUABA = 16*AGUABA_1 .
COMPUTE bbTRAB = -41*TRAB151 .
COMPUTE bbNOASI = -29*NOASI1 .
COMPUTE bbPARED = -30* PARED_1 .

IF (REFRI = 0) bbREFRI = 100 .
IF (REFRI = 1) bbREFRI = 86 .

IF (VEHI = 0) bbVEHI = 100 .
(VEHI = 1) bbVEHI = 71 .

IF (ESTGAS = 0) bbESTGAS = 100 .
IF (ESTGAS = 1) bbESTGAS = 78 .

IF (TV = 0) bbTV = 100 .
IF (TV = 1) bbTV = 79 .

EXECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
VARIABLES= bbdepen bbninos bbocupj bbesc bbpiso bbedad bbaguaba bbtrab bbnoasi bbpared
bbrefri bbvehi bbestgas bbtv
/FORMAT=NOTABLE
/STATISTICS=MEAN .

EXECUTE .
DO IF (MISSING(bbdepen) | MISSING(bbninos) | MISSING (bbocupj)| MISSING (bbesc)| MISSING (bbpiso)

| MISSING(bbrefri) | MISSING (bbvehi)| MISSING(bbestgas) |MISSING(bbtv)) .
