REGIÓN 45. VALLE DE SAN QUINTIN
FASE 8
MODELO aa

DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES= ninos mu15a49 depen trab151 ej
piso_1 ocupj_1 discap esc
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS=0.42 0.58
/STATISTICS=RAW
/HISTORY
/CLASSIFY=NONMISSING POOLED .

IF (PERCEP = 0) DEPEN = 99 .
RECODE DEPEN (99 = SYSMIS) .

COMPUTE AANINOS=-32*NINOS .
COMPUTE AAMU1549=31*MU15A49.
COMPUTE AADEPEN=-43*DEPEN.
COMPUTE AAT15=-89*TRAB151 .
COMPUTE AAEDAD=-1*ej .
COMPUTE AAPISO=15*PISO_1 .
COMPUTE AAOCUP=-56*OCUPJ_1 .
COMPUTE AAESC=5*ESC .

IF (DISCAP =0) AADISCAP = 100.
IF (DISCAP =1) AADISCAP = 52 .

FREQUENCIES
VARIABLES= AANINOS AAMU1549 AADEPEN AAT15 AAEDAD AAPISO AAOCUP AAESC AADISCAP
/FORMAT=NOTABLE
/STATISTICS=MEAN .

COMPUTE input = 0 .
EXECUTE .
DO IF (MISSING(AANINOS) | MISSING(AAMU1549) | MISSING(AADEPEN) | MISSING(AAT15)
    | MISSING(AAEDAD) | MISSING(AAPISO) | MISSING(AAOCUP) | MISSING(AAESC) | MISSING(AADISCAP)) .
RECODE
  input (0=1) .
END IF .
EXECUTE .

RECODE AANINOS (SYSMIS= -52.8854) .
RECODE AAmu1549 (SYSMIS=32.8754).
RECODE AADEPEN (SYSMIS= -86.5160) .
RECODE AAT15 (SYSMIS= -7.7545).
RECODE AAEDAD (SYSMIS= -39.4895).
RECODE AAPISO (SYSMIS= -7.2765).
RECODE AAOCUP (SYSMIS= 50.5193).
RECODE AAESC (SYSMIS= 20.6726).
RECODE AADISCAP (SYSMIS= 98.8555).

COMPUTE AAcalif = AADEPEN + AANINOS + AAmu1549 + AAT15 + AAESC + AAOCUP + AAEDAD + AAPISO + AADISCAP .
EXECUTE .

DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES=AAcalif
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS = .42 .58
/STATISTICS=raw TABLE CROSSVALID
CLASSIFY=NONMISSING POOLED .

USE ALL.
COMPUTE filter_$=(dis_1 = 0).
VARIABLE LABEL filter_$ 'dis_1 = 0 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.

FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
VARIABLES=AAcalif /FORMAT=NOTABLE
/STATISTICS=MINIMUM .

USE ALL.
COMPUTE filter_$=(dis_1 = 1).
VARIABLE LABEL filter_$ 'dis_1 = 1 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
VARIABLES=AAcalif /FORMAT=NOTABLE
/STATISTICS= MAXIMUM .

FILTER OFF.
USE ALL.
EXECUTE .

IF ((TOT_PER<=4 & ej<=24) | (TOT_PER<=2 & ej >= 65) & missing(seg_soc)) viejitos=-100 .
RECODE viejitos (SYSMIS=0) .
EXECUTE .

COMPUTE AACALIF1=AACALIF + viejitos .
EXECUTE .

IF (agua1=1 & luz=1 & piso= 1) AACALIF1=AACALIF1-100 .
EXECUTE .

COMPUTE aapobre = 0 .
EXECUTE .
DO IF (AAcalif1<7.4) .
RECODE
aapobre (0=1) .
END IF .
EXECUTE .

IF (SEG_SOC > 0 & SEG_SOC < 6) SEGURO = 1 .
IF (FAM_ENTR >= 1 & FAM_ENTR <= 7) FAM_BIEN = 1 .
IF ((SEGURO = 1 | FAM_BIEN = 1) & AAPOBRE = 1) ELIMIN = 1 .
RECODE SEGURO FAM_BIEN ELIMIN (SYSMIS = 0) .
IF (ELIMIN = 1) AAPOBRE = 0 .
EXECUTE .

FREQUENCIES
VARIABLES= aapobre lp98 elimin .
