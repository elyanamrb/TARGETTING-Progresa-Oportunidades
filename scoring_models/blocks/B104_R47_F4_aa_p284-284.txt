  /VARIABLES=depen trab151 seg_soc1 ninos vehi
  /ANALYSIS ALL
  /PRIORS EQUAL
  /STATISTICS=raw TABLE CROSSVALID
  /CLASSIFY=NONMISSING POOLED .

  VARIABLES= AADEPEN AATRAB15 AANINOS AASEGSOC AAVEHI
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

DO IF ( MISSING(AADEPEN) | MISSING(AATRAB15) | MISSING(AANINOS) |
  missing(AASEGSOC) | missing(AAVEHI)) .

  input (0=1) .
