***discriminante con la línea de pobreza

DISCRIMINANT
/GROUPS=LP98(0 1)
/VARIABLES=aacalif
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS = .42 .58
/STATISTICS=raw TABLE
/CLASSIFY=NONMISSING POOLED .

USE ALL
COMPUTE filter_$=(dis_1 = 0).
VARIABLE LABEL filter_$ 'dis_1 = 0 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS=MINIMUM .

USE ALL
COMPUTE filter_$=(dis_1 = 1).
VARIABLE LABEL filter_$ 'dis_1 = 1 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS= MAXIMUM .

FILTER OFF.
USE ALL
ΛECUTE .

***filtros

IF ((EJ < 25 & TOTPER < 5) | (EJ > 65 & EJ < 98 & TOTPER < 3)) VIEJITOS = -100 .
RECODE viejitos (SYSMIS = 0) .
EXECUTE.

COMPUTE AACALIF = AACALIF + viejitos .
EXECUTE.

***se determina la línea de corte y se obtienen los pobres

COMPUTE aapobre = 0 .
EXECUTE.
DO IF (aacalif < 374.50) .
RECODE
  aapobre (0=1) .
END IF .
EXECUTE .

IF (VEHI = 0) AAPOBRE = 0.

IF (SEG_SOC > 0 & SEG_SOC < 6) SEGURO = 1 .
IF (FAM_ENTR >= 1 & FAM_ENTR <= 7) FAM_BIEN = 1 .

IF (ELIMIN = 1) AAPOBRE = 0 .

EXECUTE .
