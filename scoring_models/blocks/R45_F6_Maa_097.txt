DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES= ninos mu15a49 depen trab151 ej
piso_1 ocupj_1 discap esc
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS=0.42 0.58
/STATISTICS=RAW
/HISTORY
/CLASSIFY=NONMISSING POOLED .

COMPUTE AANINOS=-32*NINOS .
COMPUTE AAMU1549=31*MU15A49.
COMPUTE AADEPEN=-43*DEPEN.
COMPUTE AAT15 =-89*TRAB151 .
COMPUTE AAEDAD=-1*ej .
COMPUTE AAPISO=15*PISO_1 .
COMPUTE AAOCUP=-56*OCUPJ_1 .
COMPUTE AAESC=5*ESC .

IF (DISCAP =0) AADISCAP = 100.
IF (DISCAP =1) AADISCAP = 52 .

FREQUENCIES
  VARIABLES= AANINOS AAMU1549 AADEPEN AAT15 AAEDAD AAPISO AAOCUP AAESC AADISCAP
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE input = 0 .
EXECUTE .
DO IF ( MISSING(AANINOS) | MISSING(AAMU1549) | MISSING(AADEPEN) | MISSING(AAT15)
  | MISSING(AAEDAD) | MISSING(AAPISO) | MISSING(AAOCUP) | MISSING(AAESC) | MISSING(AADISCAP) ) .
RECODE
  input (0=1) .
END IF .
EXECUTE .

RECODE AANINOS (SYSMIS=-53.02) .
RECODE AAmu1549 (SYSMIS=33).
RECODE AADEPEN (SYSMIS=-86.38) .
RECODE AAT15 (SYSMIS=-7.78).
RECODE AAEDAD (SYSMIS=-39.45).
RECODE AAPISO (SYSMIS=-7.21).
RECODE AAOCUP (SYSMIS=50.61).
RECODE AAESC (SYSMIS=20.67).
RECODE AADISCAP (SYSMIS=98.87).

COMPUTE AAcalif = AADEPEN + AANINOS + AAmu1549 + AAT15 + AAESC + AAOCUP + AAEDAD + AAPISO + AADISCAP.
EXECUTE .

DISCRIMINANT
  /GROUPS=lp98(0 1)
  /VARIABLES=AAcalif
  /ANALYSIS ALL
  /SAVE=CLASS
  /PRIORS = .42 .58
  /STATISTICS=raw TABLE CROSSVALID
  /CLASSIFY=NONMISSING POOLED .
