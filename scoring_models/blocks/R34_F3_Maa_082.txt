*/*/* FASE 3
*/*/* MODELO AA

DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES=depen ninos trab151 esc piso_1 aguaba_1
    lava estgas vhs discap ventila
/ANALYSIS ALL
/PRIORS = .42 .58
/STATISTICS=RAW TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

COMPUTE AADEPEN=-29*DEPEN .
COMPUTE AANINOS=-29*NINOS .
COMPUTE AATRAB15 = -74*TRAB151 .
COMPUTE AAESC = 3*ESC .
COMPUTE AAPISO = 7*PISO_1 .
COMPUTE AAAGUABA = 10*AGUABA_1 .

IF (DISCAP = 0) AADISCAP = 100 .
IF (DISCAP = 1) AADISCAP =72 .

IF (LAVA=0) AALAVA= 100 .
IF (LAVA=1) AALAVA= 89 .

IF (ESTGAS = 0) AAESTGAS = 100 .
IF (ESTGAS = 1) AAESTGAS = 69 .

IF (VHS = 0) AAVHS = 100 .
IF (VHS = 1) AAVHS = 68 .

IF (VENTILA = 0) AAVENTIL = 100 .
IF (VENTILA = 1) AAVENTIL = 73 .

EXECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
  VARIABLES= aadepen aaninos aatrab15 aaesc
  aapiso aaaguaba aalava aaestgas aavhs aadiscap aaventil
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE aainput = 0 .
EXECUTE .
DO IF (MISSING(aadepen) | MISSING(aaninos) | MISSING (aatrab15)
  | MISSING(aaesc) | MISSING (aapiso) | MISSING (aaaguaba)
  | MISSING(aalava) | MISSING(aaestgas) |
  MISSING(aavhs) | MISSING(aadiscap) | MISSING(aaventil)) .
RECODE
  aainput (0=1) .
END IF .
EXECUTE .

***se recodifican los valores perdidos con las medias de las variables

RECODE aadepen (SYSMIS=-71.5010) .
RECODE aaninos (SYSMIS=-47.4363) .
RECODE aatrab15 (SYSMIS=-2.2118) .
RECODE aaesc (SYSMIS=8.4146) .
RECODE aapiso (SYSMIS=.07719 ) .
RECODE aaaguaba (SYSMIS=.0004) .
RECODE aalava (SYSMIS=90.1205) .
RECODE aaestgas (SYSMIS=82.4966) .
RECODE aavhs (SYSMIS=70.0637) .
RECODE aadiscap (SYSMIS= 980931) .
RECODE aaventil (SYSMIS=75.5390) .

***se hace la calificación

COMPUTE aacalif = aadepen + aaninos + aatrab15 + aaesc +
  aapiso + aaaguaba + aalava + aaestgas + aavhs + aadiscap
  + aaventil .
EXECUTE .

***discriminante con la línea de pobreza

DISCRIMINANT
/GROUPS=LP98(0 1)

/VARIABLES=aacalif
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS = .42 .58
/STATISTICS=raw TABLE
/CLASSIFY=NONMISSING POOLED .

USE ALL.
COMPUTE filter_$=(dis_1 = 0).
VARIABLE LABEL filter_$ 'dis_1 = 0 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
VARIABLES=aacalif
/FORMAT=NOTABLE
/STATISTICS=MINIMUM .

USE ALL.
COMPUTE filter_$=(dis_1 = 1).
VARIABLE LABEL filter_$ 'dis_1 = 1 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
VARIABLES=aacalif
/FORMAT=NOTABLE
/STATISTICS= MAXIMUM .

FILTER OFF.
USE ALL.
EXECUTE .

***filtros

((EJ < 25 & TOTPER < 5) | (EJ > 65 & EJ < 98 & TOTPER < 3)) VIEJITOS = -100 .
IF (estgas = 1 & licua=1) cocina = -100 .
IF ( iv = 1 & radio = 1 ) entret = -100.

RECODE viejitos cocina entret (SYSMIS = 0) .

EXECUTE .

COMPUTE AACALIF = AACALIF + viejitos + cocina + entret .
EXECUTE.

***se determina la linea de corte y se obtienen los pobres

COMPUTE aapobre = 0 .
EXECUTE .
DO IF (aacalif < 352.435) .
RECODE
  aapobre (0=1)
END IF
EXECUTE .

IF (VEHI = 0) AAPOBRE = 0.
EXECUTE.

FREQUENCIES
VARIABLES= lp98 aapobre .
