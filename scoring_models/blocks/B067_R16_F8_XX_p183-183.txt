Región 16. Pochutla
Fase 8

***discriminante con modelo nuevo

DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES= lengua esc discap trab151 noasi1 vehi hacina
depen ninos luz licua refri tv
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS .42 .58
/STATISTICS=RAW TABLE
/CLASSIFY=NONMISSING POOLED .

***puntajes región 16

***Calificación

IF (PERCEP = 0) DEPEN = 99 .
RECODE DEPEN (99 = SYSMIS) .

COMPUTE AAESC = 7.86*ESC .
COMPUTE AATRAB = -33.075*TRAB151 .
COMPUTE AANOASI = -13.867*NOASI1 .
COMPUTE AAHACINA = -5.021*HACINA .
COMPUTE AADEPEN = -18.058*DEPEN .
COMPUTE AANINOS = -15.86*NINOS .

IF (LENGUA = 0) AALENGUA = 100 .
IF (LENGUA = 1) AALENGUA = 96 .

IF (DISCAP = 0) AADISCAP = 100 .
IF (DISCAP = 1) AADISCAP = 66 .

IF (VEHI = 0) AAVEHI = 100 .
IF (VEHI = 1) AAVEHI = 39 .

IF (LUZ = 0) AALUZ = 100 .
IF (LUZ = 1) AALUZ = 75 .

IF (LICUA = 0) AALICUA = 100 .
IF (LICUA = 1) AALICUA = 47 .

IF (REFRI = 0) AAREFRI = 100 .
IF (REFRI = 1) AAREFRI = 59 .

IF (TV = 0) AATV = 100 .
IF (TV = 1) AATV = 100 .

EXECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES

aaluz aalicua aarefri aatv

/STATISTICS=MEAN .

MISSING(aalicua) | MISSING (aarefri)| MISSING(aatv)) .
RECODE
aainput (0=1) .
END IF .
EXECUTE .
