REGIÓN 12. HUASTECA
FASE 8
MODELO aa


DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES=depen ninos trab151 hacina esc discap radio tv estgas ventila hectie1
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS .42 .58
/STATISTICS=RAW TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

IF (PERCEP = 0) DEPEN = 99 .
RECODE DEPEN (99 = SYSMIS) .

COMPUTE AADEPEN=-42*DEPEN .
COMPUTE AANINOS=-3*NINOS .
COMPUTE AATRAB15 = -124*TRAB151 .
COMPUTE AAHACINA = -6*HACINA .
COMPUTE AAESC = 8*ESC .
COMPUTE AAHECTIE = 2*HECTIE1

IF (DISCAP = 0) AADISCAP = 100 .
IF (DISCAP = 1) AADISCAP = 79 .

IF (RADIO = 0) AARADIO = 100 .
IF (RADIO = 1) AARADIO = 96 .

IF (TV = 0) AATV = 100 .
IF (TV = 1) AATV = 67 .

IF (ESTGAS=0) AAESTGAS= 100 .
IF (ESTGAS=1) AAESTGAS= 84 .

IF (VENTILA= 0) AAVENTIL= 100 .
IF (VENTILA= 1) AAVENTIL= 56 .

EXECUTE.

FREQUENCIES
VARIABLES= aadepen aaninos aatrab15 aahacina aaesc aahectie aadiscap aaradio aatv aaestgas aaventil
/FORMAT=NOTABLE
/STATISTICS=MEAN .

COMPUTE aainput = 0 .
EXECUTE .
DO IF (MISSING(aadepen) | MISSING(aaninos) | MISSING (aatrab15) | MISSING(aahacina) | MISSING(aaesc)
| MISSING(aahectie) | MISSING(aadiscap) | MISSING(aaradio) | MISSING(aatv) | MISSING(aaestgas)
| MISSING(aaventil)) .
RECODE
aainput (0=1) .
END IF .
EXECUTE .

RECODE aadepen (SYSMIS = -95.2553) .
RECODE aaninos (SYSMIS = -4.7500) .
RECODE aatrab15 (SYSMIS = -3.4488) .
RECODE aahacina (SYSMIS = -17.8581) .
RECODE aaesc (SYSMIS = 25.9664) .
RECODE aahectie (SYSMIS = 6.2492) .
RECODE aadiscap (SYSMIS = 98.7557) .
RECODE aaradio (SYSMIS = 98.7911) .
RECODE aatv (SYSMIS = 88.2480) .
RECODE aaestgas (SYSMIS = 91.8743) .
RECODE aaventil (SYSMIS = 71.8945) .

COMPUTE aacalif = aadepen + aaninos + aatrab15 + aahacina + aaesc + aahectie + aadiscap + aaradio + aatv
+ aaestgas + aaventil.
EXECUTE .

  /STATISTICS=MINIMUM .

  /STATISTICS= MAXIMUM .

EXECUTE .

   recode viejitos (SYSMIS = 0) .
