*/*/ se calculan los puntajes

COMPUTE AADEPEN=-35*DEPEN.
COMPUTE AANINOS=-11*NINOS .
COMPUTE AANA15=-18*NOASI1.
COMPUTE AAT15=-14*TRAB151 .
COMPUTE AAHACINA=-2*HACINA .
COMPUTE AAESC=6*ESC .
COMPUTE AAOCUP=23*OCUPJ_1 .
COMPUTE AAPISO=11*PISO_1 .
COMPUTE AAHECTI=2*HECTIE1 .
COMPUTE AAEDAD=1*ej .

IF (DISCAP =0) AADISCAP = 100.
IF (DISCAP =1) AADISCAP = 87 .

IF (RADIO=0) AARADIO=100 .
IF (RADIO=1) AARADIO=94 .

if (TV=0) AATV = 100.
IF (TV=1) AATV =79 .

IF (LICUA=0) AALICUA= 100.
IF (LICUA=1) AALICUA=77 .

    (ESTGAS=0) AAESTGAS=100 .
    (ESTGAS=1) AAESTGAS=70 .

IF (VEHI=0) AAVEHI=100 .
IF (VEHI=1) AAVEHI=68 .

EXECUTE.

FREQUENCIES
  VARIABLES= AADEPEN AANINOS AANA15 AAT15 AAHACINA AAESC AAOCUP AAPISO AAHECTI
    AAEDAD AADISCAP AARADIO AATV AALICUA AAESTGAS AAVEHI
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE aainput = 0 .
EXECUTE .
DO IF ( MISSING(AADEPEN) I MISSING(AANINOS) I MISSING(AANA15)
   I MISSING(AAT15) I MISSING(AAHACINA) I MISSING(AAESC)
   I MISSING(AAOCUP) I MISSING(AAPISO) I missing(AAHECTI) I missing(AAEDAD) I
     missing(AADISCAP) I missing(AARADIO) I MISSING(AATV) I MISSING(AALICUA) I
     missing(AAESTGAS) I missing(AAVEHI) ) .
RECODE
   aainput (0=1) .
END IF .
EXECUTE .

RECODE AADEPEN (SYSMIS=-100.69) .
RECODE AANINOS (SYSMIS=-20.17) .
RECODE AANA15 (SYSMIS=-3.048) .
RECODE AAT15 (SYSMIS=-7.66).
