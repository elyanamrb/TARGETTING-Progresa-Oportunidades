***Modelo

DISCRIMINANT
/GROUPS=LP98(0 1)
/VARIABLES= depen ninos noasi1 esc
agua_1 estgas vhs refri vehi lava discap
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS = .42 .58
/STATISTICS=raw TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

***Calificación

COMPUTE AADEPEN=-25*DEPEN .
COMPUTE AANINOS=-26*NINOS .
COMPUTE AANOASI=-33*NOASI1 .
COMPUTE AAESC=8*ESC .
COMPUTE AAAGUA=12*AGUA_1 .

IF (DISCAP=0) AADISCAP = 100 .
IF (DISCAP=1) AADISCAP = 86 .

IF (ESTGAS=0) AAESTGAS = 100 .
IF (ESTGAS=1) AAESTGAS = 63 .

IF (VHS=0) AAVHS= 100 .
IF (VHS=1) AAVHS= 56 .

IF (REFRI=0) AAREFRI= 100 .
IF (REFRI=1) AAREFRI= 79 .

IF (VEHI=0) AAVEHI= 100 .
IF (VEHI=1) AAVEHI= 66 .

IF (LAVA = 0) AALAVA = 100 .
IF (LAVA = 1) AALAVA = 67 .

EXECUTE .

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
VARIABLES= aadepen aaninos aanoasi aaesc
aaagua aadiscap aaestgas aavhs aarefri aavehi aalava
/FORMAT=NOTABLE
/STATISTICS=MEAN .

COMPUTE aainput = 0 .
EXECUTE .
DO IF (MISSING(aadepen) | MISSING(aaninos) | MISSING(aanoasi) | MISSING(aaesc)
| MISSING(aaagua) | MISSING(aadiscap) | MISSING(aaestgas) | MISSING(aavhs) | MISSING(aarefri) | MISSING (aavehi) |
MISSING (aalava)) .
RECODE
aainput (0=1) .
END IF .
EXECUTE .

***se recodifican los valores perdidos con las medias de las variables

RECODE aadepen (SYSMIS=-65.51) .
RECODE aaninos (SYSMIS=-38.62) .
