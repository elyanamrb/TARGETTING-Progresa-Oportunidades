***se hace la calificación

COMPUTE aacalif = aadepen + aaninos + aanoasi1+ aatrab15 + aahacina +
  aaesc + aaej + aapiso + aaaguab + aadiscap + aaestgas + aaventil +
  + aalengua + aavehi .
XECUTE .

***discriminante con la linea de pobreza

DISCRIMINANT
/GROUPS=LP98(0 1)
/VARIABLES=aacalif
/ANALYSIS ALL
/SAVE=CLASS
/PRIORS .42 .58
/STATISTICS=MEAN STDDEV UNIVF BOXM RAW CORR COV GCOV TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

USE ALL.
COMPUTE filter_$=(dis_2 = 0).
VARIABLE LABEL filter_$ 'dis_2= 0 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS=MINIMUM .

USE ALL.
COMPUTE filter_$=(dis_2 = 1).
VARIABLE LABEL filter_$ 'dis_2 = 1 (FILTER)'.
VALUE LABELS filter_$ 0 'Not Selected' 1 'Selected'.
FORMAT filter_$ (f1.0).
FILTER BY filter_$.
EXECUTE .

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS= MAXIMUM .

FILTER OFF.
USE ALL.
EXECUTE .

if ((ej<25 & TOTPER<5) | (ej>65 & TOTPER <3)) viejitos=-100 .
recode viejitos (sysmis=0) .
execute .

COMPUTE AAcalif1 = AAcalif + viejitos.
EXECUTE .

***se determina la linea de corte y se obtienen los pobres

COMPUTE aapobre = 0 .

  aapobre (0=1) .
END IF .

  VARIABLES=AAPOBRE lp98 .
