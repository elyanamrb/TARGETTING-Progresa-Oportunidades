DISCRIMINANT
  /GROUPS=lp98(0 1)
  /VARIABLES=hacina depen ninos ocupj_1 esc mu15a49 trab151 noasi1
             vhs refri vehi lava licua radio estgas tv discap
  /ANALYSIS ALL
  /SAVE=CLASS
  /PRIORS = .42 .58
  /STATISTICS=RAW TABLE crossvalid
  /CLASSIFY=NONMISSING POOLED .

*** Puntajes Región 8

IF (PERCEP = 0) DEPEN = 99.
RECODE DEPEN (99 = SYSMIS) .

COMPUTE AAHACINA= 2*HACINA.
COMPUTE AADEPEN= 46*DEPEN.
COMPUTE AANINOS= 25*NINOS.
COMPUTE AAOCUPJ= 42*OCUPJ_1 .
COMPUTE AAESC= 52*ESC .
COMPUTE AAMU15A=10*MU15A49 .
COMPUTE AATRAB= 57*TRAB151 .
COMPUTE AANOASI= 14*NOASI1 .

IF (VHS =0) AAVHS = 100.
IF (VHS =1) AAVHS = 75 .

IF (REFRI=0) AAREFRI=100 .
IF (REFRI=1) AAREFRI=85 .

if (VEHI=0) AAVEHI = 100.
IF (VEHI=1) AAVEHI =86 .

IF (LAVA=0) AALAVA= 100.
IF (LAVA=1) AALAVA=84 .

  (LICUA =0) AALICUA = 100.
  (LICUA =1) AALICUA = 90 .

IF (RADIO=0) AARADIO=100 .
IF (RADIO=1) AARADIO=94 .

if (ESTGAS=0) AAESTGAS = 100.
IF (ESTGAS=1) AAESTGAS =84 .

IF (TV=0) AATV= 100.
IF (TV=1) AATV=89 .

IF (DISCAP=0) AADISCAP= 100.
IF (DISCAP=1) AADISCAP=80 .

EXECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
  VARIABLES=AAHACINA AADEPEN AANINOS AAOCUPJ AAESC AAMU15A AATRAB AANOASI
            AAVHS AAREFRI AAVEHI AALAVA AALICUA AARADIO AAESTGAS AATV AADISCAP
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE input = 0 .
EXECUTE .
DO IF (MISSING(AAHACINA) | MISSING(AADEPEN) | MISSING(AANINOS) | MISSING(AAOCUPJ) | MISSING(AAESC)
      | MISSING(AAMU15A) | MISSING(AATRAB) | MISSING(AANOASI) | MISSING(AAVHS) | MISSING(AAREFRI)
      | MISSING(AAVEHI) | missing(AALAVA) | missing(AALICUA) | MISSING(AARADIO) | MISSING(AAESTGAS)
      | MISSING(AATV) | MISSING(AADISCAP)) .
COMPUTE
   input (0=1) .
END IF .
*EXECUTE .

***se recodifican los valores perdidos con las medias de las variables

RECODE  AAHACINA (SYSMIS=5.2940) .
RECODE  AADEPEN (SYSMIS=-103.3956) .
