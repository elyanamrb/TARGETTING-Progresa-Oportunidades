cal_agua tocadisc ventila

/SAVE=CLASS
/PRIORS = .42 .58
/STATISTICS=raw TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

***Calificación

COMPUTE AADEPEN=.52*DEPEN .
COMPUTE AANINOS=.18*NINOS .
COMPUTE AATRAB15 = .75 * TRAB151 .
COMPUTE AAOCUPJ = 8 * OCUPJ_1 .
COMPUTE AAESC = 3 * ESC .
COMPUTE AAPISO = 2 * PISO1.
COMPUTE AAAGUABA = 3*AGUABA_1 .
COMPUTE AAPARED = .13 * PARED1.
COMPUTE AAHECTI = 1 * HECTIE1 .

IF (DISCAP = 0) AADISCAP = 100 .
IF (DISCAP = 1) AADISCAP = 78 .

IF (RADIO = 0) AARADIO = 100 .
IF (RADIO = 1) AARADIO = 93 .

IF (ESTGAS = 0) AAESTGAS = 100.
IF (ESTGAS = 1) AAESTGAS = 81 .

IF (VHS = 0) AAVHS = 100 .
IF (VHS = 0) AAVHS = 80 .

IF (VEHI=0) AAVEHI= 100 .
IF (VEHI=1) AAVEHI= 78 .

IF (LAVA = 0) AALAVA = 100 .
IF (LAVA = 1) AALAVA = 83 .

IF (CAL_AGUA = 0) AACAL_AG = 100 .
IF (CAL_AGUA = 1) AACAL_AG = 90 .

IF (TOCADISC = 0) AATOCADI = 100 .
IF (TOCADISC = 1) AATOCADI = 83 .

IF (VENTILA = 0) AAVENTI = 100 .
IF (VENTILA = 1) AAVENTI = 85 .

EXECUTE .

***calcular las medias de las variables para imputar a valores perdidos

  VARIABLES= aadepen aaninos aatrab15 aaocupj aaesc aapiso aaaguaba aapared aahecti aadiscap
             aaradio aaestgas aavhs aavehi aalava aacal_ag aatocadi aaventi
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

DO IF (MISSING(aadepen) | MISSING(aaninos) | MISSING(aatrab15) | MISSING(aaocupj) | MISSING(aaesc)

| MISSING(aaestgas) | MISSING(aavhs) | MISSING(aavehi) | MISSING (aalava) | MISSING(aacal_ag) | MISSING (aatocadi)
| MISSING(aaventi) ) .

  ainput (0=1) .
END IF .
EXECUTE .
