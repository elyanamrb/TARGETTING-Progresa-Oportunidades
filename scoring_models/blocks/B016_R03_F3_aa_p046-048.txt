* LINEA DE POBREZA

*code ingmesh ingmen15(99999=sysmis) .

compute inmh15pc=(ingmesh-ingmen15)/totpers.

if (sysmis(ingmesh) I sysmis(ingmen15))inmh15pc=999.
recode inmh15pc (999=sysmis).

compute IpNVA=0.
if (inmh15pc <= 286.06) IpNVA=1.
if (sysmis(ingmesh))IpNVA=99.
recode IpNVA (99=sysmis).

* INDICE DE DEPENDENCIA

IF (PERCEP~=0) depen= (totpers-percep)/percep.
IF (PERCEP=0) DEPEN= TOTPERS.

* DISCAPACITADOS

RENAME VARIABLES (P431=DISCAP) .

*HACINAMIENTO

RECODE P61 (99=SYSMIS).
compute hacina= totpers/p61.

IF (hacina <= 1) hh1 = 1 .
RECODE hh1 (MISSING=0).
IF (hacina > 1 & hacina<=2) hh2 = 1 .
RECODE hh2 (MISSING=0) .
IF (hacina > 2 & hacina<=3) hh3 = 1 .
RECODE hh3 (MISSING=0) .
IF (hacina > 3 & hacina<=4) hh4 = 1 .
RECODE hh4 (MISSING=0) .
IF (hacina > 4 & hacina<=5) hh5 = 1 .
RECODE hh5 (MISSING=0).
IF (hacina > 5 ) hh6 = 1.
RECODE hh6 (MISSING=0) .

IF (SYSMIS(hacina)) hh1=99.
IF (SYSMIS(hacina)) hh2=99.
IF (SYSMIS(hacina)) hh3=99.
IF (SYSMIS(hacina)) hh4=99.
IF (SYSMIS(hacina)) hh5=99.
IF (SYSMIS(hacina)) hh6=99.

RECODE hh1 hh2 hh3 hh4 hh5 hh6 (99=SYSMIS) .

compute esc=escjefe .
if (esccony > escjefe & esccony <99) esc= esccony.

IF (ESC=0) ESCJ1 = 1 .

RECODE ESCJ2 (MISSING=0) .
IF (ESC=6) ESCJ3 = 1 .
RECODE ESCJ3 (MISSING=0) .
