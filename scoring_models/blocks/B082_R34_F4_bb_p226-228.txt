DISCRIMINANT
/GROUPS=lp98(0 1)
/VARIABLES=depen ninos trab151 esc
    lava vhs discap ventila
/ANALYSIS ALL
/PRIORS = .41 .59
/STATISTICS=RAW TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

****PUNTAJES

COMPUTE BBDEPEN=-36*DEPEN .
COMPUTE BBNINOS=-29*NINOS .
COMPUTE BBTRAB15 = -53*TRAB151 .
COMPUTE BBESC = 5*ESC .

IF (DISCAP = 0) BBDISCAP = 100 .
IF (DISCAP = 1) BBDISCAP =56 .

IF (LAVA=0) BBLAVA= 100 .
IF (LAVA=1) BBLAVA= 85 .

IF (VHS = 0) BBVHS = 100 .
IF (VHS = 1) BBVHS = 98 .

IF (VENTILA = 0) BBVENTIL = 100 .
IF (VENTILA = 1) BBVENTIL = 71 .

EXECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
  VARIABLES= BBdepen BBninos BBtrab15 BBesc
  BBlava BBvhs BBdiscap BBventil
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE BBinput = 0 .
EXECUTE .
DO IF (MISSING(BBdepen) | MISSING(BBninos) | MISSING (BBtrab15)
  | MISSING(BBesc) | MISSING(BBBinput) | MISSING (aaaguaba)
  | MISSING(BBlava) | MISSING(BBBblava) |
  MISSING(BBvhs) | MISSING(BBdiscap) | MISSING(BBventil)) .
RECODE
  BBinput (0=1) .
END IF .
EXECUTE .

***se recodifican los valores perdidos con las medias de las variables

RECODE BBdepen (SYSMIS=-89.82) .
RECODE BBninos (SYSMIS=-37.68) .
RECODE BBtrab15 (SYSMIS=-1.51) .
RECODE BBesc (SYSMIS=30.91) .
RECODE BBlava (SYSMIS=88.40) .
RECODE BBvhs (SYSMIS=98.26) .
RECODE BBdiscap (SYSMIS=97.33) .
RECODE BBventil (SYSMIS=78.10) .

***se hace la calificación

COMPUTE BBcalif = BBdepen + BBninos + BBtrab15 + BBesc +
  BBlava+ BBvhs + BBdiscap
  + BBventil .
EXECUTE .
