/VARIABLES=depen ninos trab151 esc piso_1 aguaba_1
   lava estgas vhs discap ventila
/ANALYSIS ALL
/PRIORS = .42 .58
/STATISTICS=RAW TABLE CROSSVALID
/CLASSIFY=NONMISSING POOLED .

*****************************************

GET
  FILE='h:\4_5y6fase\punteo\reg_34\reg34j_8.sav'
EXECUTE .

COMPUTE AADEPEN=-29*DEPEN .
COMPUTE AANINOS=-29*NINOS .
COMPUTE AATRAB15 = -74*TRAB151 .
COMPUTE AAESC = 3*ESC .
COMPUTE AAPISO = 7*PISO_1.
COMPUTE AAAGUABA = 10*AGUABA_1 .

IF (DISCAP = 0) AADISCAP = 100 .
IF (DISCAP = 1) AADISCAP =72 .

IF (LAVA=0) AALAVA= 100 .
IF (LAVA=1) AALAVA= 89 .

IF (ESTGAS = 0) AAESTGAS = 100 .
IF (ESTGAS = 1) AAESTGAS = 69 .

IF (VHS = 0) AAVHS = 100 .
IF (VHS = 1) AAVHS = 68 .

IF (VENTILA = 0) AAVENTIL = 100 .
IF (VENTILA = 1) AAVENTIL = 73 .

ΛECUTE.

***calcular las medias de las variables para imputar a valores perdidos

FREQUENCIES
  VARIABLES= aadepen aaninos aatrab15 aaesc
  aapiso aaaguaba aalava aaestgas aavhs aadiscap aaventil
  /FORMAT=NOTABLE
  /STATISTICS=MEAN .

COMPUTE ainput = 0 .
EXECUTE .
DO IF (MISSING(aadepen) | MISSING(aaninos) | MISSING (aatrab15)
  | MISSING(aaesc) | MISSING (aapiso) | MISSING (aaaguaba)
  | MISSING(aalava) | MISSING(aaestgas) |
  MISSING(aavhs) | MISSING(aadiscap) | MISSING(aaventil))
RECODE
  ainput (0=1) .
END IF .
EXECUTE .

***se recodifican los valores perdidos con las medias de las variables

RECODE aadepen (SYSMIS=-74.6002) .
RECODE aaninos (SYSMIS=-36.3813) .
RECODE aatrab15 (SYSMIS=-2.6599) .
RECODE aaesc (SYSMIS=10.2287) .
RECODE aapiso (SYSMIS=-.8214) .
RECODE aaaguaba (SYSMIS=-2.5881) .
RECODE aalava (SYSMIS=91.0958) .
RECODE aaestgas (SYSMIS=93.6690) .
RECODE aavhs (SYSMIS=72.9728) .
RECODE aadiscap (SYSMIS=98.8169) .
RECODE aaventil (SYSMIS=76.6707) .

*se hace la calificación

COMPUTE aacalif = aadepen + aaninos + aatrab15 + aaesc +
  aapiso + aaaguaba + aalava + aaestgas + aavhs + aadiscap
  + aaventil .

COMPUTE filter_$=(dis_1 = 0).

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS=MINIMUM .

FILTER BY filter_$.
EXECUTE .

FREQUENCIES
  VARIABLES=aacalif
  /FORMAT=NOTABLE
  /STATISTICS= MAXIMUM .

USE ALL.
EXECUTE .

***filtros

IF ((EJ < 25 & TOTPER < 5) | (EJ > 65 & EJ < 98 & TOTPER < 3)) VIEJITOS = -100 .
